第 11 章 在 Unity/URP 里落地：一个可运行的 Toon Shader

把前面这些层落成工程，最小的结构是：一个 UniversalForward Pass 算颜色，一个 Outline Pass 画描边，再加可选的 DepthOnly/ShadowCaster。材质属性和贴图声明放在一个共享的 HLSL 文件里，两个 Pass 都 include，避免 SRP Batcher 因为 CBUFFER 不一致而失效。

共享的输入文件

// ToonInput.hlsl
#ifndef TOON_INPUT_INCLUDED
#define TOON_INPUT_INCLUDED

CBUFFER_START(UnityPerMaterial)
    float4 _BaseMap_ST;
    float4 _BaseColor;
    float4 _ShadowColor;
    float4 _RimColor;
    float4 _ToonSpecColor;
    float4 _OutlineColor;
    float  _ShadowThreshold;
    float  _ShadowSoftness;
    float  _RimWidth;
    float  _RimPower;
    float  _OutlineWidth;
CBUFFER_END

TEXTURE2D(_BaseMap);   SAMPLER(sampler_BaseMap);
TEXTURE2D(_LightMap);  SAMPLER(sampler_LightMap);
TEXTURE2D(_RampTex);   SAMPLER(sampler_RampTex);

#endif

把所有材质属性放进同一个 CBUFFER_START(UnityPerMaterial)，是 SRP Batcher 能合批的前提。两个 Pass 如果各写一份不一样的 CBUFFER，Unity 会认为材质布局不一致，合批直接失效。

主 Pass：颜色与光照

Pass
{
    Name "ForwardLit"
    Tags { "LightMode" = "UniversalForward" }

    HLSLPROGRAM
    #pragma vertex vert
    #pragma fragment frag
    #pragma multi_compile _ _MAIN_LIGHT_SHADOWS
    #pragma multi_compile _ _MAIN_LIGHT_SHADOWS_CASCADE
    #pragma multi_compile _ _SHADOWS_SOFT

    #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
    #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"
    #include "ToonInput.hlsl"

    struct Attributes
    {
        float4 positionOS : POSITION;
        float3 normalOS   : NORMAL;
        float2 uv         : TEXCOORD0;
        float4 color      : COLOR;
    };

    struct Varyings
    {
        float4 positionCS  : SV_POSITION;
        float2 uv          : TEXCOORD0;
        float3 normalWS    : TEXCOORD1;
        float3 positionWS  : TEXCOORD2;
        float4 color       : TEXCOORD4;
        float4 shadowCoord : TEXCOORD3;
    };

    Varyings vert (Attributes v)
    {
        Varyings o;
        VertexPositionInputs p = GetVertexPositionInputs(v.positionOS.xyz);
        VertexNormalInputs n = GetVertexNormalInputs(v.normalOS);
        o.positionCS = p.positionCS;
        o.positionWS = p.positionWS;
        o.normalWS = n.normalWS;
        o.uv = TRANSFORM_TEX(v.uv, _BaseMap);
        o.color = v.color;
        o.shadowCoord = GetShadowCoord(p);
        return o;
    }

    half4 frag (Varyings i) : SV_Target
    {
        float3 N = normalize(i.normalWS);
        float3 V = normalize(GetWorldSpaceViewDir(i.positionWS));
        Light mainLight = GetMainLight(i.shadowCoord);
        float3 L = mainLight.direction;

        float halfLambert = dot(N, L) * 0.5 + 0.5;
        float shadow = smoothstep(_ShadowThreshold,
                                  _ShadowThreshold + _ShadowSoftness,
                                  halfLambert);

        float3 baseColor = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, i.uv).rgb;
        float3 ramp = SAMPLE_TEXTURE2D(_RampTex, sampler_RampTex,
                                       float2(halfLambert, 0.5)).rgb;
        float3 color = ramp * baseColor * _BaseColor.rgb;
        color = lerp(color * _ShadowColor.rgb, color, shadow);

        float3 H = normalize(L + V);
        float spec = pow(saturate(dot(N, H)), 32.0);
        spec = step(0.9, spec);
        color += _ToonSpecColor.rgb * spec;

        float rim = 1.0 - saturate(dot(N, V));
        rim = smoothstep(1.0 - _RimWidth, 1.0, rim);
        color += _RimColor.rgb * rim;

        return half4(color, 1.0);
    }
    ENDHLSL
}

Outline Pass：平滑法线从切线来

Pass
{
    Name "Outline"
    Tags { "LightMode" = "SRPDefaultUnlit" }
    Cull Front
    ZWrite On

    HLSLPROGRAM
    #pragma vertex vert
    #pragma fragment frag
    #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
    #include "ToonInput.hlsl"

    struct Attributes
    {
        float4 positionOS : POSITION;
        float4 tangentOS  : TANGENT;
        float4 color      : COLOR;
    };

    struct Varyings { float4 positionCS : SV_POSITION; };

    Varyings vert (Attributes v)
    {
        Varyings o;
        float3 smoothNormal = v.tangentOS.xyz;
        float3 posOS = v.positionOS.xyz
                     + smoothNormal * _OutlineWidth * 0.01 * v.color.a;
        o.positionCS = TransformObjectToHClip(posOS);
        return o;
    }

    half4 frag (Varyings i) : SV_Target
    {
        return _OutlineColor;
    }
    ENDHLSL
}

这里的 tangentOS.xyz 不是切线，而是上一章 C# 工具写进去的平滑法线。如果模型没有导入切线，或者导入时被重新计算，描边就会退回硬边效果。

Shader Graph 还是 HLSL？

维度Shader GraphHLSL
上手速度快，节点可视化慢，需要熟悉 URP 库
多 Pass / 描边不方便，通常要配合手写 Pass直接写，控制力强
分支/关键字Boolean Keyword 会产生变体可用位掩码/手写分支
调试节点预览方便Frame Debugger / RenderDoc 更直接
适合原型、简单角色、美术自制量产角色、复杂多 Pass、性能敏感

Unity 视角：先把主 Pass 写对，再考虑后处理。URP 的 GetMainLight(shadowCoord) 只在定义了阴影关键字时才有阴影；如果场景没有主光阴影，shadowCoord 仍然可以传，但结果不受影响。还有，Outline Pass 的 LightMode 用 SRPDefaultUnlit 可以避免被 UniversalForward 重复调用；如果你用 RenderObjects 单独画描边，则要确保它只对目标 Layer 生效。

本章要点：一个最小 URP Toon Shader = 共享输入 + UniversalForward + Outline。CBUFFER 要一致，平滑法线走 Tangent，描边宽度走顶点 Alpha，后处理用 Renderer Feature。先把这条链跑通，再往上加 SDF、MatCap、PBR。
